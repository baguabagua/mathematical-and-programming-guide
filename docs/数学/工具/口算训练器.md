下面是 python 版口算训练器的代码。假设保存到 `a.py`，则使用命令是 `python a.py`

```python
import tkinter as tk
from tkinter import messagebox, ttk
import random

class MentalArithmeticTrainer:
    def __init__(self, root):
        self.root = root
        self.root.title("口算能力训练器")
        self.root.geometry("650x600")
        self.root.resizable(False, False)

        # 变量
        self.operator = tk.StringVar(value="add")
        self.digits1 = tk.IntVar(value=2)
        self.digits2 = tk.IntVar(value=2)
        self.questions_count = tk.IntVar(value=10)
        self.questions = []          # 存储题目 (text, answer)
        self.answer_entries = []     # 存储Entry组件

        # 设置界面
        self.create_settings_frame()
        self.create_questions_frame()

    def create_settings_frame(self):
        """设置区域：题型、数位、题目数量"""
        frame = tk.LabelFrame(self.root, text="训练设置", padx=10, pady=10, font=("微软雅黑", 10, "bold"))
        frame.pack(fill="x", padx=10, pady=5)

        # 题型选择
        tk.Label(frame, text="题型:").grid(row=0, column=0, sticky="w", padx=5, pady=5)
        op_frame = tk.Frame(frame)
        op_frame.grid(row=0, column=1, sticky="w")
        tk.Radiobutton(op_frame, text="加法 (+)", variable=self.operator, value="add").pack(side="left")
        tk.Radiobutton(op_frame, text="减法 (-)", variable=self.operator, value="sub").pack(side="left", padx=10)
        tk.Radiobutton(op_frame, text="乘法 (×)", variable=self.operator, value="mul").pack(side="left")

        # 数位设置（根据题型动态改变提示）
        self.digits_frame = tk.Frame(frame)
        self.digits_frame.grid(row=1, column=0, columnspan=2, sticky="w", padx=5, pady=5)

        self.label_digits1 = tk.Label(self.digits_frame, text="第一个数最多位数:")
        self.label_digits1.pack(side="left")
        self.digits1_spin = tk.Spinbox(self.digits_frame, from_=1, to=5, width=5, textvariable=self.digits1)
        self.digits1_spin.pack(side="left", padx=5)

        self.label_digits2 = tk.Label(self.digits_frame, text="第二个数最多位数:")
        self.label_digits2.pack(side="left", padx=(10,0))
        self.digits2_spin = tk.Spinbox(self.digits_frame, from_=1, to=5, width=5, textvariable=self.digits2)
        self.digits2_spin.pack(side="left", padx=5)

        # 绑定题型变化，动态调整标签文字
        self.operator.trace_add("write", self.update_digits_label)

        # 题目数量
        tk.Label(frame, text="题目数量:").grid(row=2, column=0, sticky="w", padx=5, pady=5)
        tk.Spinbox(frame, from_=1, to=50, width=8, textvariable=self.questions_count).grid(row=2, column=1, sticky="w")

        # 生成按钮
        tk.Button(frame, text="生成题目", command=self.generate_questions, bg="#4CAF50", fg="white",
                  font=("微软雅黑", 10)).grid(row=3, column=0, columnspan=2, pady=10)

    def update_digits_label(self, *args):
        """根据题型改变第二个数的标签提示"""
        op = self.operator.get()
        if op == "mul":
            self.label_digits2.config(text="第二个数最多位数:")
        else:
            self.label_digits2.config(text="第二个数最多位数:")

    def create_questions_frame(self):
        """题目区域：动态显示题目和输入框"""
        self.canvas = tk.Canvas(self.root)
        scrollbar = tk.Scrollbar(self.root, orient="vertical", command=self.canvas.yview)
        self.scrollable_frame = tk.Frame(self.canvas)

        self.scrollable_frame.bind(
            "<Configure>",
            lambda e: self.canvas.configure(scrollregion=self.canvas.bbox("all"))
        )
        self.canvas.create_window((0, 0), window=self.scrollable_frame, anchor="nw")
        self.canvas.configure(yscrollcommand=scrollbar.set)

        self.canvas.pack(side="left", fill="both", expand=True, padx=10, pady=5)
        scrollbar.pack(side="right", fill="y")

        # 提交按钮区域
        self.submit_frame = tk.Frame(self.root)
        self.submit_frame.pack(fill="x", padx=10, pady=10)
        self.submit_btn = tk.Button(self.submit_frame, text="提交并批改", command=self.check_answers,
                                    bg="#2196F3", fg="white", font=("微软雅黑", 11), height=1, width=15)
        self.submit_btn.pack()

        # 结果显示区域
        self.result_label = tk.Label(self.root, text="", font=("微软雅黑", 10), fg="blue")
        self.result_label.pack(pady=5)

    def generate_questions(self):
        """生成随机题目"""
        # 清空旧题目
        for widget in self.scrollable_frame.winfo_children():
            widget.destroy()
        self.answer_entries.clear()
        self.questions.clear()

        op = self.operator.get()
        d1 = self.digits1.get()
        d2 = self.digits2.get()
        count = self.questions_count.get()

        for i in range(count):
            # 生成随机数
            num1 = self.random_number(d1)
            num2 = self.random_number(d2)

            # 减法确保结果非负（被减数≥减数）
            if op == "sub" and num1 < num2:
                num1, num2 = num2, num1

            # 根据题型计算答案
            if op == "add":
                answer = num1 + num2
                text = f"{i+1}.  {num1} + {num2} = ?"
            elif op == "sub":
                answer = num1 - num2
                text = f"{i+1}.  {num1} - {num2} = ?"
            else:  # mul
                answer = num1 * num2
                text = f"{i+1}.  {num1} × {num2} = ?"

            self.questions.append((text, answer))

            # 创建题目标签和输入框
            frame = tk.Frame(self.scrollable_frame)
            frame.pack(fill="x", pady=5)

            label = tk.Label(frame, text=text, font=("微软雅黑", 11), anchor="w", width=25)
            label.pack(side="left", padx=5)

            entry = tk.Entry(frame, width=10, font=("微软雅黑", 11))
            entry.pack(side="left", padx=10)
            self.answer_entries.append(entry)

        # 清空之前的结果
        self.result_label.config(text=f"已生成 {count} 道题，请作答后提交。")

    def random_number(self, max_digits):
        """生成一个最多max_digits位的整数（1到10^max_digits-1）"""
        if max_digits == 1:
            return random.randint(1, 9)   # 避免0
        lower = 10 ** (max_digits - 1)
        upper = 10 ** max_digits - 1
        return random.randint(lower, upper)

    def check_answers(self):
        """检查所有答案并给出批改结果"""
        if not self.questions:
            messagebox.showwarning("提示", "请先生成题目！")
            return

        correct_count = 0
        results = []

        for idx, (question_text, correct_answer) in enumerate(self.questions):
            entry = self.answer_entries[idx]
            user_input = entry.get().strip()

            if user_input == "":
                results.append(f"第{idx+1}题未作答 ❌")
                continue

            try:
                user_answer = int(user_input)
                if user_answer == correct_answer:
                    correct_count += 1
                    results.append(f"第{idx+1}题 正确 ✅")
                else:
                    results.append(f"第{idx+1}题 错误 ❌ (正确答案: {correct_answer})")
            except ValueError:
                results.append(f"第{idx+1}题 输入无效 (请输入整数) ❌")

        # 显示批改结果
        total = len(self.questions)
        score = (correct_count / total) * 100 if total > 0 else 0
        summary = f"得分: {correct_count}/{total}  ({score:.1f}分)"

        # 在界面上显示详细结果（可滚动）
        result_window = tk.Toplevel(self.root)
        result_window.title("批改结果")
        result_window.geometry("400x500")
        result_window.resizable(False, False)

        tk.Label(result_window, text=summary, font=("微软雅黑", 12, "bold"), fg="green").pack(pady=10)

        text_widget = tk.Text(result_window, wrap="word", font=("微软雅黑", 10))
        scroll = tk.Scrollbar(result_window, command=text_widget.yview)
        text_widget.configure(yscrollcommand=scroll.set)
        text_widget.pack(side="left", fill="both", expand=True, padx=10, pady=5)
        scroll.pack(side="right", fill="y")

        for line in results:
            text_widget.insert("end", line + "\n")
        text_widget.config(state="disabled")

        # 更新主窗口提示
        self.result_label.config(text=f"批改完成！ {summary}")

if __name__ == "__main__":
    root = tk.Tk()
    app = MentalArithmeticTrainer(root)
    root.mainloop()
```
